package com.tropical.tvlauncher.utils

import com.google.gson.JsonObject
import com.google.gson.JsonParser
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.OkHttpClient
import okhttp3.Request
import java.text.SimpleDateFormat
import java.util.Calendar
import java.util.Date
import java.util.Locale

data class Prayer(
    val name: String,
    val time: String,
    val dateTime: Long
)

object PrayerTimeUtils {
    private val client = OkHttpClient()
    private const val BASE_URL = "https://api.myquran.com/v2/sholat/jadwal"

    // ID Kota Karawang (MyQuran)
    private const val KARAWANG_CITY_ID = "1609"

    suspend fun getNearestPrayerForKarawang(): Pair<String, String> = withContext(Dispatchers.IO) {
        try {
            val sdf = SimpleDateFormat("yyyy/MM/dd", Locale("id", "ID"))
            val today = sdf.format(Date())
            val url = "$BASE_URL/$KARAWANG_CITY_ID/$today"

            val request = Request.Builder().url(url).get().build()
            val response = client.newCall(request).execute()
            val body = response.body?.string() ?: return@withContext Pair("Imsak", "--:--")

            val json = JsonParser.parseString(body).asJsonObject
            val data = json.getAsJsonObject("data")
            val jadwal = data.getAsJsonObject("jadwal")

            val now = Calendar.getInstance()
            val timeFormat = SimpleDateFormat("HH:mm", Locale.getDefault())

            val list = listOf(
                Prayer("Imsak", jadwal.get("imsak").asString, toMillis(jadwal.get("imsak").asString)),
                Prayer("Subuh", jadwal.get("subuh").asString, toMillis(jadwal.get("subuh").asString)),
                Prayer("Terbit", jadwal.get("terbit").asString, toMillis(jadwal.get("terbit").asString)),
                Prayer("Dhuha", jadwal.get("dhuha").asString, toMillis(jadwal.get("dhuha").asString)),
                Prayer("Dzuhur", jadwal.get("dzuhur").asString, toMillis(jadwal.get("dzuhur").asString)),
                Prayer("Ashar", jadwal.get("ashar").asString, toMillis(jadwal.get("ashar").asString)),
                Prayer("Maghrib", jadwal.get("maghrib").asString, toMillis(jadwal.get("maghrib").asString)),
                Prayer("Isya", jadwal.get("isya").asString, toMillis(jadwal.get("isya").asString))
            )

            val currentMillis = now.timeInMillis
            val next = list.firstOrNull { it.dateTime > currentMillis } ?: list.first()

            val diffMillis = next.dateTime - currentMillis
            val diffMinutes = if (diffMillis > 0) (diffMillis / 60000).toInt() else 0
            val hoursLeft = diffMinutes / 60
            val minsLeft = diffMinutes % 60

            val countdown = when {
                hoursLeft > 0 && minsLeft > 0 -> "${hoursLeft}j ${minsLeft}m lagi"
                hoursLeft > 0 -> "${hoursLeft}j lagi"
                else -> "${minsLeft}m lagi"
            }

            Pair(next.name, "${next.time} | $countdown")
        } catch (e: Exception) {
            Pair("Dzuhur", "--:-- | cek koneksi")
        }
    }

    private fun toMillis(timeStr: String): Long {
        val parts = timeStr.split(":")
        val cal = Calendar.getInstance()
        cal.set(Calendar.HOUR_OF_DAY, parts[0].toInt())
        cal.set(Calendar.MINUTE, parts[1].toInt())
        cal.set(Calendar.SECOND, 0)
        cal.set(Calendar.MILLISECOND, 0)
        return cal.timeInMillis
    }
}

class AppListAdapter(
    private val context: Context,
    private val apps: List<ResolveInfo>
) : RecyclerView.Adapter<AppListAdapter.AppViewHolder>() {

    inner class AppViewHolder(itemView: View) : RecyclerView.ViewHolder(itemView) {
        val icon: ImageView = itemView.findViewById(R.id.ivAppIcon)
        val name: TextView = itemView.findViewById(R.id.tvAppName)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): AppViewHolder {
        val view = LayoutInflater.from(context).inflate(R.layout.item_app, parent, false)
        return AppViewHolder(view)
    }

    override fun onBindViewHolder(holder: AppViewHolder, position: Int) {
        val app = apps[position]
        val pm = context.packageManager
        holder.icon.setImageDrawable(app.activityInfo.loadIcon(pm))
        holder.name.text = app.loadLabel(pm)

        holder.itemView.setOnClickListener {
            val intent = pm.getLaunchIntentForPackage(app.activityInfo.packageName)
            intent?.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            context.startActivity(intent)
        }

        // Fokus untuk navigasi remote TV
        holder.itemView.isFocusable = true
        holder.itemView.isFocusableInTouchMode = true
    }

    override fun getItemCount(): Int = apps.size
}

        bgImageView = findViewById(R.id.bgImageView)
        tvPrayerInfo = findViewById(R.id.tvPrayerInfo)
        ivWifi = findViewById(R.id.ivWifi)
        ivSettings = findViewById(R.id.ivSettings)
        tvDayDate = findViewById(R.id.tvDayDate)
        tvClock = findViewById(R.id.tvClock)
        rvAppsRow1 = findViewById(R.id.rvAppsRow1)
        rvAppsRow2 = findViewById(R.id.rvAppsRow2)
    }

    private fun setupBackground() {
        showNextBackground()
        bgHandler.postDelayed(bgRunnable, 5 * 60 * 1000)
    }

    private fun showNextBackground() {
        if (backgroundList.isEmpty()) return
        currentBgIndex = (currentBgIndex + 1) % backgroundList.size
        Glide.with(this)
            .load(backgroundList[currentBgIndex])
            .centerCrop()
            .into(bgImageView)
    }

    private fun setupClock() {
        updateClock()
        clockHandler.postDelayed(clockRunnable, 1000)
    }

    private fun updateClock() {
        val now = Date()
        val dayDateFormat = SimpleDateFormat("EEEE, dd MMMM yyyy", Locale("id", "ID"))
        val clockFormat = SimpleDateFormat("HH:mm", Locale.getDefault())
        tvDayDate.text = dayDateFormat.format(now)
        tvClock.text = clockFormat.format(now)
    }

    private fun setupWifi() {
        updateWifiStatus()
        val filter = IntentFilter(ConnectivityManager.CONNECTIVITY_ACTION)
        registerReceiver(wifiReceiver, filter)
    }

    private fun updateWifiStatus() {
        val cm = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        var connected = false
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            val network = cm.activeNetwork
            val caps = cm.getNetworkCapabilities(network)
            connected = caps?.hasTransport(NetworkCapabilities.TRANSPORT_WIFI) == true ||
                    caps?.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET) == true
        } else {
            val networkInfo = cm.activeNetworkInfo
            connected = networkInfo?.isConnected == true
        }
        if (connected) {
            ivWifi.setImageResource(R.drawable.ic_wifi)
            ivWifi.contentDescription = getString(R.string.wifi_connected)
        } else {
            ivWifi.setImageResource(R.drawable.ic_wifi_off)
            ivWifi.contentDescription = getString(R.string.wifi_disconnected)
        }
    }

    private fun setupSettings() {
        ivSettings.setOnClickListener {
            val intent = Intent(Settings.ACTION_SETTINGS)
            intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            startActivity(intent)
        }
        ivSettings.setOnFocusChangeListener { v, hasFocus ->
            v.scaleX = if (hasFocus) 1.15f else 1.0f
            v.scaleY = if (hasFocus) 1.15f else 1.0f
        }
    }

    private fun loadPrayerTimes() {
        prayerScope.launch {
            val result = withContext(Dispatchers.IO) {
                PrayerTimeUtils.getNearestPrayerForKarawang()
            }
            tvPrayerInfo.text = getString(R.string.prayer_nearest, result.first, result.second.split(" | ")[0], result.second.split(" | ").getOrElse(1) { "" })
        }
        // Refresh tiap 60 detik biar countdown tetap akurat
        val timer = Timer()
        timer.scheduleAtFixedRate(object : TimerTask() {
            override fun run() {
                prayerScope.launch {
                    val result = withContext(Dispatchers.IO) {
                        PrayerTimeUtils.getNearestPrayerForKarawang()
                    }
                    runOnUiThread {
                        tvPrayerInfo.text = getString(R.string.prayer_nearest, result.first, result.second.split(" | ")[0], result.second.split(" | ").getOrElse(1) { "" })
                    }
                }
            }
        }, 60000, 60000)
    }

    private fun loadInstalledApps() {
        val intent = Intent(Intent.ACTION_MAIN, null)
        intent.addCategory(Intent.CATEGORY_LAUNCHER)
        val pm = packageManager
        val apps = pm.queryIntentActivities(intent, 0)
            .sortedBy { it.loadLabel(pm).toString().lowercase(Locale.getDefault()) }

        val list1 = apps.take(5)
        val list2 = apps.drop(5).take(5)

        rvAppsRow1.layoutManager = GridLayoutManager(this, 5)
        rvAppsRow1.adapter = AppListAdapter(this, list1)

        rvAppsRow2.layoutManager = GridLayoutManager(this, 5)
        rvAppsRow2.adapter = AppListAdapter(this, list2)
    }

    override fun onDestroy() {
        super.onDestroy()
        bgHandler.removeCallbacks(bgRunnable)
        clockHandler.removeCallbacks(clockRunnable)
        try {
            unregisterReceiver(wifiReceiver)
        } catch (e: Exception) {
        }
    }
